// Helper membaca data Sheet
function getSheetData(sheetName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data.shift();
  return data.map(row => {
    let obj = {};
    headers.forEach((header, index) => { 
      let key = String(header).trim();
      obj[key] = row[index]; 
    });
    return obj;
  });
}

// Helper konversi URL Google Drive ke Direct Image Link
function convertDriveUrlToDirect(url) {
  if (!url) return '';
  let strUrl = String(url).trim();
  const match = strUrl.match(/\/d\/([a-zA-Z0-9_-]+)/);
  if (match && match[1]) {
    return 'https://lh3.googleusercontent.com/d/' + match[1];
  }
  return strUrl;
}

// Mengambil daftar kelas master utama
function getDaftarKelasMaster() {
  initSheetSiswa();
  const dataSiswa = getSheetData('Siswa');
  let kelasList = dataSiswa.map(s => String(s.Kelas).trim()).filter(Boolean);

  if (kelasList.length === 0) {
    const users = getSheetData('Pengguna');
    kelasList = users
      .filter(u => String(u.Is_Walas) === 'true' || String(u.Is_Walas) === 'TRUE')
      .map(u => String(u.Kelas_Walas).trim())
      .filter(Boolean);
  }

  if (kelasList.length === 0) {
    const dataWalasSheet = getSheetData('Data_Wali_Kelas');
    kelasList = dataWalasSheet.map(s => String(s.Kelas).trim()).filter(Boolean);
  }

  return [...new Set(kelasList)].sort();
}

function simpanSiswa(data) {
  const sheet = initSheetSiswa();
  const allData = sheet.getDataRange().getValues();

  const newNis = String(data.nis || '').trim();
  const oldNis = String(data.oldNis || '').trim();

  // Jika Edit Siswa berdasarkan Primary Key (NIS)
  if (oldNis) {
    for (let i = 1; i < allData.length; i++) {
      const currentNis = String(allData[i][0]).trim();
      if (currentNis === oldNis) {
        sheet.getRange(i + 1, 1, 1, 4).setValues([[data.nis, data.nisn, data.nama, data.kelas]]);
        return "Data siswa berhasil diperbarui!";
      }
    }
  }

  // Cek duplikasi NIS sebagai Primary Key
  for (let i = 1; i < allData.length; i++) {
    const currentNis = String(allData[i][0]).trim();
    if (newNis && currentNis === newNis) {
      return "NIS sudah terdaftar sebagai Primary Key! Gunakan NIS lain.";
    }
  }

  sheet.appendRow([data.nis, data.nisn, data.nama, data.kelas]);
  return "Data siswa berhasil ditambahkan!";
}

function hapusSiswa(nis) {

      return "Data siswa berhasil dihapus!";
    }
  }
  return "Data siswa tidak ditemukan!";
}

function importSiswaBatch(dataArray) {
  if (!dataArray || dataArray.length === 0) return "Data impor siswa kosong!";
  const sheet = initSheetSiswa();
  const allData = sheet.getDataRange().getValues();

  dataArray.forEach(item => {
    const nis = String(item.NIS || item.nis || '').trim();
    const nisn = String(item.NISN || item.nisn || '').trim();
    const nama = item.Nama_Siswa || item.Nama || item.nama || '';
    const kelas = item.Kelas || item.kelas || '';

    let rowIndex = -1;
    for (let i = 1; i < allData.length; i++) {
      if (nis && String(allData[i][0]).trim() === nis) {
        rowIndex = i + 1;
        break;
      }
    }

    if (rowIndex > 0) {
      sheet.getRange(rowIndex, 1, 1, 4).setValues([[nis, nisn, nama, kelas]]);
    } else {
      sheet.appendRow([nis, nisn, nama, kelas]);
    }
  });

  return "Data Siswa berhasil diimpor!";
}

function processLogin(username, password) {
  const users = getSheetData('Pengguna');
  const user = users.find(u => String(u.Username).trim() === String(username).trim() && String(u.Password).trim() === String(password).trim());
  
  if (user) {
    return {
      success: true,
      user: {
        id: user.ID,
        nama: user.Nama,
        role: user.Role,
        username: user.Username,
        mapelAkses: user.Mapel_Akses ? String(user.Mapel_Akses).split(',').map(s => s.trim()) : [],
        kelasAkses: user.Kelas_Akses ? String(user.Kelas_Akses).split(',').map(s => s.trim()) : [],
        isWalas: String(user.Is_Walas) === 'true' || String(user.Is_Walas) === 'TRUE',
        kelasWalas: String(user.Kelas_Walas || '').trim()
      }
    };
  }

  initSheetWaliMurid();
  const waliMuridList = getSheetData('Wali_Murid');
  const wali = waliMuridList.find(w => String(w.Username).trim() === String(username).trim() && String(w.Password).trim() === String(password).trim());

  if (wali) {
    return {
      success: true,
      user: {
        id: 'WM-' + (wali.NIS || new Date().getTime()),
        nama: wali.Nama_Orang_Tua || 'Wali Murid',
        role: 'Wali Murid',
        username: wali.Username,
        nisSiswa: wali.NIS,
        nisnSiswa: wali.NISN,
        namaSiswa: wali.Nama_Siswa,
        kelasSiswa: wali.Kelas,
        mapelAkses: [],
        kelasAkses: [],
        isWalas: false,
        kelasWalas: ''
      }
    };
  }

  return { success: false, message: 'Username atau Password salah!' };
}

function simpanPengguna(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Pengguna') || ss.insertSheet('Pengguna');
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID', 'Username', 'Password', 'Nama', 'Role', 'Mapel_Akses', 'Kelas_Akses', 'Is_Walas', 'Kelas_Walas']);
  }
  
  const mapelStr = Array.isArray(data.mapelAkses) ? data.mapelAkses.join(',') : '';
  const kelasStr = Array.isArray(data.kelasAkses) ? data.kelasAkses.join(',') : '';
  const allData = sheet.getDataRange().getValues();

  if (data.id) {
    for (let i = 1; i < allData.length; i++) {
      if (String(allData[i][0]) === String(data.id)) {
        sheet.getRange(i + 1, 2, 1, 8).setValues([[
          data.username,
          data.password,
          data.nama,
          data.role,
          mapelStr,
          kelasStr,
          data.isWalas,
          data.kelasWalas
        ]]);
        return "Data pengguna berhasil diperbarui!";
      }
    }
  }

  sheet.appendRow([
    'USR-' + new Date().getTime(),
    data.username,
    data.password,
    data.nama,
    data.role,
    mapelStr,
    kelasStr,
    data.isWalas,
    data.kelasWalas
  ]);
  return "Pengguna baru berhasil ditambahkan!";
}

function generateWaliMuridAutomatic() {
  initSheetSiswa();
  initSheetWaliMurid();

  const siswaList = getSheetData('Siswa');
  if (siswaList.length === 0) return "Data siswa masih kosong!";

  const sheetWM = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Wali_Murid');
  const wmExisting = getSheetData('Wali_Murid');

  let addedCount = 0;

  siswaList.forEach(s => {
    const nis = String(s.NIS || '').trim();
    if (!nis) return;

    const exists = wmExisting.some(w => String(w.NIS).trim() === nis);
    if (!exists) {
      const nisn = s.NISN || '';
      const namaSiswa = s.Nama_Siswa || s.Nama || '';
      const kelas = s.Kelas || '';
      const namaOrtu = 'Orang Tua / Wali ' + namaSiswa;
      const username = 'parent_' + nis;
      const password = 'wm' + nis;

      sheetWM.appendRow([nis, nisn, namaSiswa, kelas, namaOrtu, username, password]);
      addedCount++;
    }
  });

  return `Berhasil membuat ${addedCount} akun Wali Murid baru secara otomatis!`;
}

function simpanKehadiranBatch(dataArray) {

  const allData = sheet.getDataRange().getValues();
  if (allData.length === 0) return "Sheet Data_Wali_Kelas kosong!";

  const headers = allData[0].map(h => String(h).trim());
  let colNis = headers.indexOf('NIS');
  let colNisn = headers.indexOf('NISN');
  let colNama = headers.indexOf('Nama');
  let colKelas = headers.indexOf('Kelas');
  let colSakit = headers.indexOf('Sakit_(Hari)');
  let colIzin = headers.indexOf('Izin_(Hari)');
  let colAlpa = headers.indexOf('Alpa_(Hari)');

  if (colSakit === -1 || colIzin === -1 || colAlpa === -1) {
    return "Struktur kolom pada Sheet Data_Wali_Kelas tidak sesuai!";
  }

  dataArray.forEach(item => {
    let rowIndex = -1;
    for (let i = 1; i < allData.length; i++) {
      const rowNis = String(allData[i][colNis] || '').trim();

      sheet.getRange(rowIndex, colIzin + 1).setValue(item.izin);

    } else {
      let newRow = new Array(headers.length).fill('');
      if (colNis !== -1) newRow[colNis] = item.nis || '';
      if (colNisn !== -1) newRow[colNisn] = item.nisn || '';
      if (colNama !== -1) newRow[colNama] = item.namaSiswa || '';

}

function hitungPredikatUmum(nilai) {
  const n = Number(nilai);
  if (n >= 90) return 'A';
  if (n >= 80) return 'B';
  if (n >= 70) return 'C';
  return 'D';
}

function hitungPredikatLughoh(n1, n2, n3) {
  const total = Number(n1 || 0) + Number(n2 || 0) + Number(n3 || 0);
  if (total >= 50) return 'Sangat Baik (A)';
  if (total >= 35) return 'Baik (B)';
  if (total >= 20) return 'Cukup (C)';
  return 'Perlu Bimbingan (D)';
}

function simpanNilaiUmumBatch(dataArray) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Nilai_Umum') || ss.insertSheet('Nilai_Umum');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['NIS', 'NISN', 'Nama_Siswa', 'Kode_Mapel', 'Materi_CP', 'Nilai', 'Predikat', 'Capaian']);
  }

  const existingData = sheet.getDataRange().getValues();

  dataArray.forEach(item => {
    let rowIndex = -1;
    for (let i = 1; i < existingData.length; i++) {
      const rowNis = String(existingData[i][0]).trim();
      const rowKode = String(existingData[i][3]).trim();

      if (item.nis && rowNis === String(item.nis).trim() && rowKode === String(item.kodeMapel).trim()) {
        rowIndex = i + 1;
        break;
      }
    }

    const predikat = hitungPredikatUmum(item.nilai);

    if (rowIndex > 0) {
      sheet.getRange(rowIndex, 5, 1, 4).setValues([[
        item.materiCP, item.nilai, predikat, item.capaian
      ]]);
    } else {
      sheet.appendRow([
        item.nis, item.nisn, item.namaSiswa, item.kodeMapel, 
        item.materiCP, item.nilai, predikat, item.capaian
      ]);
    }
  });
  return "Nilai Mapel Umum berhasil disimpan/diperbarui!";
}

function simpanNilaiLughohBatch(dataArray) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Nilai_ShobahulLughoh') || ss.insertSheet('Nilai_ShobahulLughoh');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['NIS', 'NISN', 'Nama_Siswa', 'Bahasa', 'Jumlah_Mufradat', 'Jumlah_Muhadatsah', 'Jumlah_Mahfudzot', 'Predikat', 'Vocabulary', 'Conversation', 'Wise_Word']);
  }

  const existingData = sheet.getDataRange().getValues();

  dataArray.forEach(item => {
    let rowIndex = -1;
    const itemBahasa = item.bahasa || 'Bahasa Arab';

    for (let i = 1; i < existingData.length; i++) {
      const rowNis = String(existingData[i][0]).trim();
      const rowBahasa = String(existingData[i][3] || 'Bahasa Arab').trim();

      if (item.nis && rowNis === String(item.nis).trim() && 
          (rowBahasa === itemBahasa || (itemBahasa === 'English' && (rowBahasa === 'English' || rowBahasa === 'Bahasa Inggris')))) {
        rowIndex = i + 1;
        break;
      }
    }

    if (itemBahasa === 'English' || itemBahasa === 'Bahasa Inggris') {
      const voc = item.vocabulary !== undefined ? item.vocabulary : '';
      const conv = item.conversation !== undefined ? item.conversation : '';
      const wise = item.wiseWord !== undefined ? item.wiseWord : '';
      const predText = item.predikat !== undefined ? item.predikat : '';

      if (rowIndex > 0) {
        sheet.getRange(rowIndex, 4).setValue('English');
        sheet.getRange(rowIndex, 9).setValue(voc);
        sheet.getRange(rowIndex, 10).setValue(conv);
        sheet.getRange(rowIndex, 11).setValue(wise);
        sheet.getRange(rowIndex, 8).setValue(predText);
      } else {
        sheet.appendRow([
          item.nis, item.nisn, item.namaSiswa, 'English',
          '', '', '', predText, voc, conv, wise
        ]);
      }
    } else {
      const val1 = item.jumlahMufradat || 0;
      const val2 = item.jumlahMuhadatsah || 0;
      const val3 = item.jumlahMahfudzot || 0;
      const pred = hitungPredikatLughoh(val1, val2, val3);

      if (rowIndex > 0) {
        sheet.getRange(rowIndex, 4, 1, 5).setValues([[
          'Bahasa Arab', val1, val2, val3, pred
        ]]);
      } else {
        sheet.appendRow([
          item.nis, item.nisn, item.namaSiswa, 'Bahasa Arab',
          val1, val2, val3, pred, '', '', ''
        ]);
      }
    }
  });
  return "Nilai Shobahul Lughoh berhasil disimpan/diperbarui!";
}

function getLegerByKelas(kelas) {
  const siswaList = getSiswaByKelas(kelas);
  const nilaiUmum = getSheetData('Nilai_Umum');
  const mapelList = getSheetData('Mapel');

  let leger = siswaList.map(siswa => {
    let namaSiswa = siswa.Nama_Siswa || siswa.Nama;
    let row = { NIS: siswa.NIS, NISN: siswa.NISN, Nama: namaSiswa, NilaiMapel: {}, Total: 0, RataRata: 0 };
    let totalNilai = 0;
    let countMapel = 0;

    mapelList.forEach(m => {
      let rec = nilaiUmum.find(n => 
        String(n.NIS).trim() === String(siswa.NIS).trim() && 
        String(n.Kode_Mapel).trim() === String(m.Kode_Mapel).trim()
      );
      let score = rec ? Number(rec.Nilai) : 0;
      row.NilaiMapel[m.Kode_Mapel] = score;
      totalNilai += score;
      if (score > 0) countMapel++;
    });

    row.Total = totalNilai;
    row.RataRata = countMapel > 0 ? (totalNilai / countMapel).toFixed(2) : 0;
    return row;
  });

  return { mapelList: mapelList, legerData: leger };
}

  ) || {};

  ) || {};
